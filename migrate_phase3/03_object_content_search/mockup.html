CREATE VIEW dbo.v_customer_orders
AS
SELECT 
    c.id AS customer_id, 
    c.name AS customer_name,
    o.id AS order_id, 
    o.amount, 
    o.status, 
    o.created_at
FROM dbo.customers c
JOIN dbo.orders o ON o.customer_id = c.id
WHERE o.status = 'active';
